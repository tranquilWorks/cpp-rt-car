cmake_minimum_required(VERSION 3.20)
project(rtfw_host_lifecycle LANGUAGES C CXX)
if(NOT TARGET rtfw::runtime)
    find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS runtime)
endif()
if(NOT (WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux") OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Host lifecycle kit requires 64-bit Linux or Windows")
endif()
add_library(rtfw_host_lifecycle_plugin SHARED plugin.c)
target_compile_features(rtfw_host_lifecycle_plugin PRIVATE c_std_11)
target_include_directories(rtfw_host_lifecycle_plugin PRIVATE
    $<TARGET_PROPERTY:rtfw::runtime,INTERFACE_INCLUDE_DIRECTORIES>)
if(WIN32)
    target_compile_definitions(rtfw_host_lifecycle_plugin PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
set_target_properties(rtfw_host_lifecycle_plugin PROPERTIES PREFIX "" OUTPUT_NAME host_lifecycle_module)
add_executable(rtfw_host_lifecycle_demo main.cpp)
target_compile_features(rtfw_host_lifecycle_demo PRIVATE cxx_std_20)
target_link_libraries(rtfw_host_lifecycle_demo PRIVATE rtfw::runtime ${CMAKE_DL_LIBS})
set_target_properties(rtfw_host_lifecycle_demo PROPERTIES OUTPUT_NAME host_lifecycle)
add_dependencies(rtfw_host_lifecycle_demo rtfw_host_lifecycle_plugin)
enable_testing()
add_test(NAME host_adapter_host_lifecycle_module COMMAND rtfw_host_lifecycle_demo
    $<TARGET_FILE:rtfw_host_lifecycle_plugin>)
set_tests_properties(host_adapter_host_lifecycle_module PROPERTIES TIMEOUT 60)
