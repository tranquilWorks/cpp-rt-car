cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(rtfw_telemetry LANGUAGES CXX)
    enable_testing()
endif()
if(NOT TARGET rtfw::runtime)
    find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS runtime)
endif()
add_library(rtfw_telemetry_kit STATIC telemetry.cpp)
target_include_directories(rtfw_telemetry_kit PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(rtfw_telemetry_kit PUBLIC rtfw::runtime)
target_compile_features(rtfw_telemetry_kit PUBLIC cxx_std_20)
if(WIN32)
    target_sources(rtfw_telemetry_kit PRIVATE etw.cpp)
    target_link_libraries(rtfw_telemetry_kit PUBLIC advapi32)
endif()
if(COMMAND rtfw_enable_project_warnings)
    rtfw_enable_project_warnings(rtfw_telemetry_kit)
endif()
add_executable(rtfw_telemetry_example main.cpp)
target_link_libraries(rtfw_telemetry_example PRIVATE rtfw_telemetry_kit)
add_test(NAME telemetry_example COMMAND rtfw_telemetry_example)
set_tests_properties(telemetry_example PROPERTIES TIMEOUT 30)
# This source-kit target is never exported or installed as a default SDK target.
