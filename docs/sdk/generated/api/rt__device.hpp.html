  540 struct DeviceBackendHandle {
  541     std::uint64_t value = invalid_device_handle;
  542 
  543     constexpr DeviceBackendHandle() noexcept = default;
  544     explicit constexpr DeviceBackendHandle(
  545         std::uint64_t encoded) noexcept
  546         : value(encoded) {}
  547     constexpr DeviceBackendHandle(
  548         std::uint32_t owner,
  549         std::uint32_t index) noexcept
  550         : value(
  551               (static_cast<std::uint64_t>(owner) << 32u) |
  552               static_cast<std::uint64_t>(index)) {}
  553 
  554     [[nodiscard]] constexpr bool valid() const noexcept {
  555         return value != invalid_device_handle &&
  556                (static_cast<std::uint32_t>(value) &
  557                 device_buffer_kind_bit) == 0;
  558     }
  559     [[nodiscard]] constexpr std::uint32_t owner() const noexcept {
  560         return static_cast<std::uint32_t>(value >> 32u);
  561     }
  562     [[nodiscard]] constexpr std::uint32_t index() const noexcept {
  563         return static_cast<std::uint32_t>(value) &
  564                device_handle_index_mask;
  565     }
  566 
  567     friend constexpr bool operator==(
  568         DeviceBackendHandle,
  569         DeviceBackendHandle) noexcept = default;
  570 };
