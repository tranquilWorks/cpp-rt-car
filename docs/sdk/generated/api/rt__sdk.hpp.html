   19 template <auto Function, typename State>
   20 concept Callback =
   21     (std::is_member_function_pointer_v<decltype(Function)> ||
   22      (std::is_pointer_v<decltype(Function)> && std::is_function_v<std::remove_pointer_t<decltype(Function)>>)) &&
   23     !std::is_same_v<std::integral_constant<decltype(Function), Function>,
   24                     std::integral_constant<decltype(Function), nullptr>> &&
   25     !std::is_const_v<State> && !std::is_volatile_v<State> &&
   26     requires(State& state, const CallbackContext& context) {
   27     { std::invoke(Function, state, context) } noexcept -> std::same_as<CallbackResult>;
   28 };

   30 // State must remain at the same address until checked stop; temporaries cannot bind.
   31 // Both free functions and member functions are supported. No callable is stored.
   32 template <auto Function, typename State> requires Callback<Function, State>
   33 [[nodiscard]] CallbackRegistration callback(std::string_view name, State& state) noexcept {
   34     return {name, [](void* user, const CallbackContext& context) noexcept {
   35         return std::invoke(Function, *static_cast<State*>(user), context);
   36     }, std::addressof(state)};
   37 }

   78     std::size_t total_bytes = 0;

   81 // Pure arithmetic, not Runtime configuration validation. Output is unchanged on
   82 // failure. Zero bytes/count are valid; alignment must be a nonzero power of two.
   83 [[nodiscard]] constexpr Status aligned_capacity(std::size_t bytes, std::size_t alignment,
   84                                                 std::size_t count, AlignedCapacity& out) noexcept {
   85     if (alignment == 0 || (alignment & (alignment - 1)) != 0) return Status::invalid_argument;
   86     constexpr auto maximum = std::numeric_limits<std::size_t>::max();
   87     if (bytes > maximum - (alignment - 1)) return Status::capacity_exceeded;
   88     const auto stride = (bytes + alignment - 1) & ~(alignment - 1);
   89     if (count != 0 && stride > maximum / count) return Status::capacity_exceeded;
   90     out = {stride, stride * count};
   91     return Status::ok;
   92 }

  100 // Partial storage arithmetic only: excludes trace and all control bytes. Always
  101 // configure/finalize and inspect MemoryPlan for the authoritative budget result.
  102 // Host-adapter queues have different ownership and are deliberately unsupported.
  103 [[nodiscard]] constexpr Status estimate_native_storage(const RuntimeConfig& config,
  104     std::size_t phases, NativeStorageEstimate& out) noexcept {
  105     if (config.executor_policy != ExecutorPolicy::static_deterministic &&
  106         config.executor_policy != ExecutorPolicy::bounded_throughput) return Status::invalid_config;
  107     if (phases > config.callback_capacity || phases > config.task_scratch_slots)
  108         return Status::capacity_exceeded;
  109     NativeStorageEstimate candidate;
  110     auto status = aligned_capacity(config.scratch_bytes, config.scratch_alignment, phases,
  111                                    candidate.phase_scratch);
  112     if (status != Status::ok) return status;
  113     status = aligned_capacity(config.task_scratch_bytes, config.scratch_alignment,
  114                               config.task_scratch_slots, candidate.task_scratch);
  115     if (status != Status::ok) return status;
  116     if (config.worker_count == 0 || config.executor_queue_capacity == 0) return Status::invalid_config;
  117     if (config.executor_queue_capacity > std::numeric_limits<std::size_t>::max() / config.worker_count)
  118         return Status::capacity_exceeded;
  119     candidate.queue_slots = config.worker_count * config.executor_queue_capacity;
  120     out = candidate;
  121     return Status::ok;
  122 }

  124 // Construct after successful finalize, before start. Runtime and every borrowed
  125 // owner must outlive this guard and remain at the same address. Control-thread use
  126 // only. Prefer explicit close(); a failed close keeps the guard armed for retry.
  127 // Destruction attempts stop once, then terminates on failure instead of permitting
  128 // borrowed owners to be destroyed while Runtime still retains them.
  129 class CheckedStopGuard final {
  130 public:
  131     explicit CheckedStopGuard(Runtime& runtime) noexcept : runtime_(runtime) {}
  132     CheckedStopGuard(const CheckedStopGuard&) = delete;
  133     CheckedStopGuard& operator=(const CheckedStopGuard&) = delete;
  134     CheckedStopGuard(CheckedStopGuard&&) = delete;
  135     CheckedStopGuard& operator=(CheckedStopGuard&&) = delete;
  136     ~CheckedStopGuard() noexcept {
  137         if (close() != Status::ok) std::terminate();
  138     }
  139     [[nodiscard]] Status close() noexcept {
  140         if (!armed_) return Status::ok;
  141         const auto result = runtime_.stop();
  142         if (result == Status::ok) armed_ = false;
  143         return result;
  144     }
  145     [[nodiscard]] bool armed() const noexcept { return armed_; }
  146 private:
  147     Runtime& runtime_;
  148     bool armed_ = true;
  149 };

  187 // Inputs must not overlap output. Render immediately: Runtime::last_error() is a
  188 // borrowed view that can change on the next Runtime operation. No allocation/I/O.
  189 [[nodiscard]] inline DiagnosticResult render_error(std::span<char> output, std::string_view operation,
  190     Status status, std::string_view detail = {}) noexcept {
  191     DiagnosticResult result;
  192     auto append = [&](std::string_view text) noexcept {
  193         for (char character : text) {
  194             if (!output.empty() && result.written_bytes < output.size() - 1)
  195                 output[result.written_bytes++] = character;
  196             ++result.required_bytes;
  197         }
  198     };
  199     std::array<char, 12> number{};
  200     const auto converted = std::to_chars(number.data(), number.data() + number.size(),
  201                                          static_cast<std::int32_t>(status));
  202     append(operation); append(": "); append(status_message(status)); append(" (");
  203     append({number.data(), static_cast<std::size_t>(converted.ptr - number.data())}); append(")");
  204     if (!detail.empty()) { append(": "); append(detail); }
  205     append("; "); append(recovery_hint(status));
  206     ++result.required_bytes;
  207     if (!output.empty()) output[result.written_bytes] = '\0';
  208     result.truncated = result.required_bytes > output.size();
  209     return result;
  210 }
