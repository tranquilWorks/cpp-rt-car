   66 template <typename T>
   67 concept LiveControlCodecShape = requires(
   68     const T& input,
   69     T& output,
   70     std::span<
   71         std::byte,
   72         static_cast<std::size_t>(LiveControlTypeTraits<T>::encoded_extent)>
   73         encoded,
   74     std::span<
   75         const std::byte,
   76         static_cast<std::size_t>(LiveControlTypeTraits<T>::encoded_extent)>
   77         canonical) {
   78     { LiveControlTypeTraits<T>::application_type_identity } ->
   79         std::convertible_to<std::uint32_t>;
   80     { LiveControlTypeTraits<T>::application_schema_version } ->
   81         std::convertible_to<std::uint32_t>;
   82     { LiveControlTypeTraits<T>::update_kind } ->
   83         std::convertible_to<LiveControlUpdateKind>;
   84     { LiveControlTypeTraits<T>::encoded_extent } ->
   85         std::convertible_to<std::size_t>;
   86     { LiveControlTypeTraits<T>::validate(input) } noexcept ->
   87         std::same_as<bool>;
   88     { LiveControlTypeTraits<T>::encode(input, encoded) } noexcept ->
   89         std::same_as<bool>;
   90     { LiveControlTypeTraits<T>::decode(canonical, output) } noexcept ->
   91         std::same_as<bool>;
   92 };

   94 [[nodiscard]] inline bool live_control_typed_options_valid(
   95     std::size_t payload_bytes,
   96     LiveControlTypedRecordOptions options) noexcept {
   97     const auto alignment = options.payload_alignment;
   98     return alignment != 0 &&
   99         alignment <= live_control_payload_alignment_limit &&
  100         (alignment & (alignment - 1u)) == 0 &&
  101         (payload_bytes % alignment) == 0 &&
  102         options.policy_flags ==
  103             live_control_payload_canonical_little_endian;
  104 }

  116 template <typename T>
  117 concept LiveControlFixedType =
  118     detail::LiveControlCodecShape<T> &&
  119     !std::is_pointer_v<T> &&
  120     !std::is_array_v<T> &&
  121     std::is_standard_layout_v<T> &&
  122     std::is_trivially_copyable_v<T> &&
  123     std::is_nothrow_default_constructible_v<T> &&
  124     std::is_nothrow_copy_assignable_v<T> &&
  125     static_cast<std::uint32_t>(
  126         LiveControlTypeTraits<T>::application_type_identity) != 0 &&
  127     static_cast<std::uint32_t>(
  128         LiveControlTypeTraits<T>::application_schema_version) != 0 &&
  129     detail::live_control_update_kind_valid(
  130         static_cast<LiveControlUpdateKind>(

  132     static_cast<std::size_t>(LiveControlTypeTraits<T>::encoded_extent) != 0 &&

  134         live_control_payload_bytes_limit - live_control_typed_envelope_bytes;

  145 [[nodiscard]] inline LiveControlTypedStatus inspect_live_control_typed_envelope(
  146     std::span<const std::byte> payload,
  147     LiveControlTypedEnvelopeMetadata& output) noexcept {
  148     if (payload.size() < live_control_typed_envelope_bytes ||
  149         payload.size() > live_control_payload_bytes_limit ||
  150         payload.size() > std::numeric_limits<std::uint32_t>::max()) {
  151         return LiveControlTypedStatus::malformed_envelope;
  152     }
  153 
  154     LiveControlTypedEnvelopeMetadata candidate;
  155     std::uint32_t kind = 0;
  156     std::uint32_t reserved0 = 0;
  157     std::uint32_t reserved1 = 0;
  158     if (!load_u32_le(payload, 0, candidate.magic) ||
  159         !load_u32_le(payload, 4, candidate.envelope_version) ||
  160         !load_u32_le(payload, 8, candidate.total_encoded_bytes) ||
  161         !load_u32_le(payload, 12, candidate.application_type_identity) ||
  162         !load_u32_le(payload, 16, candidate.application_schema_version) ||
  163         !load_u32_le(payload, 20, kind) ||
  164         !load_u32_le(payload, 24, reserved0) ||
  165         !load_u32_le(payload, 28, reserved1) ||
  166         candidate.magic != live_control_typed_envelope_magic ||
  167         candidate.envelope_version != live_control_typed_envelope_version ||
  168         candidate.total_encoded_bytes != payload.size() ||
  169         candidate.application_type_identity == 0 ||
  170         candidate.application_schema_version == 0 ||
  171         kind < static_cast<std::uint32_t>(
  172             LiveControlUpdateKind::scenario_parameters) ||
  173         kind > static_cast<std::uint32_t>(LiveControlUpdateKind::clear_fault) ||
  174         reserved0 != 0 || reserved1 != 0) {
  175         return LiveControlTypedStatus::malformed_envelope;
  176     }
  177     candidate.update_kind = static_cast<LiveControlUpdateKind>(kind);
  178     output = candidate;
  179     return LiveControlTypedStatus::ok;
  180 }

  182 template <LiveControlFixedType T>
  183 [[nodiscard]] LiveControlTypedStatus encode_live_control_typed_payload(
  184     const T& value,
  185     LiveControlTypedPayload<T>& output) noexcept {
  186     if (!LiveControlTypeTraits<T>::validate(value)) {
  187         return LiveControlTypedStatus::codec_rejected;
  188     }
  189 
  190     std::fill(output.begin(), output.end(), std::byte{0});
  191     const auto type_identity = static_cast<std::uint32_t>(
  192         LiveControlTypeTraits<T>::application_type_identity);
  193     const auto schema_version = static_cast<std::uint32_t>(
  194         LiveControlTypeTraits<T>::application_schema_version);
  195     const auto kind = static_cast<std::uint32_t>(
  196         LiveControlTypeTraits<T>::update_kind);
  197     const auto total = static_cast<std::uint32_t>(output.size());
  198     if (!store_u32_le(output, 0, live_control_typed_envelope_magic) ||
  199         !store_u32_le(output, 4, live_control_typed_envelope_version) ||
  200         !store_u32_le(output, 8, total) ||
  201         !store_u32_le(output, 12, type_identity) ||
  202         !store_u32_le(output, 16, schema_version) ||
  203         !store_u32_le(output, 20, kind) ||
  204         !store_u32_le(output, 24, 0) ||
  205         !store_u32_le(output, 28, 0)) {
  206         return LiveControlTypedStatus::malformed_envelope;
  207     }
  208     std::span<
  209         std::byte,
  210         static_cast<std::size_t>(LiveControlTypeTraits<T>::encoded_extent)>
  211         body{output.data() + live_control_typed_envelope_bytes,
  212              static_cast<std::size_t>(
  213                  LiveControlTypeTraits<T>::encoded_extent)};
  214     return LiveControlTypeTraits<T>::encode(value, body)
  215         ? LiveControlTypedStatus::ok
  216         : LiveControlTypedStatus::codec_rejected;
  217 }

  219 template <LiveControlFixedType T>
  220 [[nodiscard]] LiveControlTypedStatus decode_live_control_typed_payload(
  221     const LiveControlUpdateRecord& record,
  222     std::span<const std::byte> payload,
  223     T& output) noexcept {
  224     constexpr auto expected_extent = live_control_typed_payload_extent<T>;
  225     constexpr auto expected_kind = static_cast<LiveControlUpdateKind>(
  226         LiveControlTypeTraits<T>::update_kind);
  227     if (record.schema_version != live_control_schema_version ||
  228         record.record_size != sizeof(LiveControlUpdateRecord) ||
  229         record.payload_bytes != expected_extent ||
  230         payload.size() != expected_extent ||
  231         record.policy_flags !=
  232             live_control_payload_canonical_little_endian) {
  233         return LiveControlTypedStatus::malformed_envelope;
  234     }
  235     if (record.update_kind != expected_kind) {
  236         return LiveControlTypedStatus::kind_mismatch;
  237     }
  238     if (record.payload_digest != live_control_payload_digest(payload)) {
  239         return LiveControlTypedStatus::digest_mismatch;
  240     }
  241 
  242     LiveControlTypedEnvelopeMetadata envelope;
  243     const auto inspection = inspect_live_control_typed_envelope(
  244         payload, envelope);
  245     if (inspection != LiveControlTypedStatus::ok) {
  246         return inspection;
  247     }
  248     if (envelope.application_type_identity != static_cast<std::uint32_t>(
  249             LiveControlTypeTraits<T>::application_type_identity)) {
  250         return LiveControlTypedStatus::type_mismatch;
  251     }
  252     if (envelope.application_schema_version != static_cast<std::uint32_t>(
  253             LiveControlTypeTraits<T>::application_schema_version)) {
  254         return LiveControlTypedStatus::schema_mismatch;
  255     }
  256     if (envelope.update_kind != expected_kind) {
  257         return LiveControlTypedStatus::kind_mismatch;
  258     }
  259 
  260     std::span<
  261         const std::byte,
  262         static_cast<std::size_t>(LiveControlTypeTraits<T>::encoded_extent)>
  263         body{payload.data() + live_control_typed_envelope_bytes,
  264              static_cast<std::size_t>(
  265                  LiveControlTypeTraits<T>::encoded_extent)};
  266     T candidate{};
  267     if (!LiveControlTypeTraits<T>::decode(body, candidate) ||
  268         !LiveControlTypeTraits<T>::validate(candidate)) {
  269         return LiveControlTypedStatus::codec_rejected;
  270     }
  271     output = candidate;
  272     return LiveControlTypedStatus::ok;
  273 }

  277 template <LiveControlFixedType T>
  278 [[nodiscard]] LiveControlTypedStatus make_live_control_update_base(
  279     const LiveControlProducerHandle& handle,
  280     std::uint64_t producer_sequence,
  281     const T& value,
  282     LiveControlTypedRecordOptions options,
  283     LiveControlTypedPayload<T>& payload,
  284     LiveControlUpdateRecord& output) noexcept {
  285     if (!handle.valid()) {
  286         return LiveControlTypedStatus::invalid_handle;
  287     }
  288     if (producer_sequence == 0 ||
  289         producer_sequence == std::numeric_limits<std::uint64_t>::max()) {
  290         return LiveControlTypedStatus::invalid_argument;
  291     }
  292     if (payload.size() != live_control_typed_payload_extent<T>) {
  293         return LiveControlTypedStatus::invalid_argument;
  294     }
  295     if (!live_control_typed_options_valid(payload.size(), options)) {
  296         return LiveControlTypedStatus::invalid_options;
  297     }
  298 
  299     const auto encoded = encode_live_control_typed_payload(value, payload);
  300     if (encoded != LiveControlTypedStatus::ok) {
  301         return encoded;
  302     }
  303 
  304     LiveControlUpdateRecord candidate;
  305     candidate.runtime_id = handle.runtime_id;
  306     candidate.configuration_generation = handle.configuration_generation;
  307     candidate.mailbox_identity = handle.mailbox_identity;
  308     candidate.producer_identity = handle.producer_identity;
  309     candidate.producer_sequence = producer_sequence;
  310     candidate.payload_digest = live_control_payload_digest(payload);
  311     candidate.payload_bytes = static_cast<std::uint32_t>(payload.size());
  312     candidate.payload_alignment = options.payload_alignment;
  313     candidate.policy_flags = options.policy_flags;
  314     candidate.update_kind = static_cast<LiveControlUpdateKind>(
  315         LiveControlTypeTraits<T>::update_kind);
  316     output = candidate;
  317     return LiveControlTypedStatus::ok;
  318 }

  322 template <LiveControlFixedType T>
  323 [[nodiscard]] LiveControlTypedStatus make_live_control_host_update(
  324     const LiveControlProducerHandle& handle,
  325     std::uint64_t producer_sequence,
  326     std::uint64_t target_frame_index,
  327     const T& value,
  328     LiveControlTypedPayload<T>& payload,
  329     LiveControlUpdateRecord& output,
  330     LiveControlTypedRecordOptions options = {}) noexcept {
  331     if (target_frame_index == std::numeric_limits<std::uint64_t>::max()) {
  332         return LiveControlTypedStatus::invalid_target;
  333     }
  334     LiveControlUpdateRecord candidate;
  335     const auto result = detail::make_live_control_update_base(
  336         handle,
  337         producer_sequence,
  338         value,
  339         options,
  340         payload,
  341         candidate);
  342     if (result != LiveControlTypedStatus::ok) {
  343         return result;
  344     }
  345     candidate.target_frame_index = target_frame_index;
  346     candidate.target_kind = LiveControlTargetKind::host_frame;
  347     output = candidate;
  348     return LiveControlTypedStatus::ok;
  349 }
