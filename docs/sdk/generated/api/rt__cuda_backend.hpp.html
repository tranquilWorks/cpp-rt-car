  212 [[nodiscard]] inline bool cuda_kernel_add_buffer_argument(
  213     CudaKernelLaunch& launch,
  214     std::uint8_t submission_buffer_index) noexcept {
  215     if (launch.argument_count >= launch.arguments.size()) {
  216         return false;
  217     }
  218     auto& argument = launch.arguments[launch.argument_count++];
  219     argument = {};
  220     argument.kind = static_cast<std::uint8_t>(
  221         CudaKernelArgumentKind::buffer_address);
  222     argument.buffer_index = submission_buffer_index;
  223     return true;
  224 }

  226 [[nodiscard]] inline bool cuda_kernel_add_scalar_argument(
  227     CudaKernelLaunch& launch,
  228     std::span<const std::byte> value) noexcept {
  229     if (launch.argument_count >= launch.arguments.size() ||
  230         launch.scalar_data_size > launch.scalar_data.size() ||
  231         value.empty() ||
  232         value.size() > sizeof(std::uint64_t) ||
  233         value.size() >
  234             launch.scalar_data.size() - launch.scalar_data_size) {
  235         return false;
  236     }
  237     auto& argument = launch.arguments[launch.argument_count++];
  238     argument = {};
  239     argument.kind =
  240         static_cast<std::uint8_t>(CudaKernelArgumentKind::scalar);
  241     argument.scalar_offset = launch.scalar_data_size;
  242     argument.scalar_size = static_cast<std::uint8_t>(value.size());
  243     std::memcpy(
  244         launch.scalar_data.data() + launch.scalar_data_size,
  245         value.data(),
  246         value.size());
  247     launch.scalar_data_size = static_cast<std::uint8_t>(
  248         launch.scalar_data_size + value.size());
  249     return true;
  250 }
