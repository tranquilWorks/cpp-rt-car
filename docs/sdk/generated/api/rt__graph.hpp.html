   17 // Handles are instance-local and remain stable for the lifetime of a Runtime.
   18 // Owner and kind components reject foreign-runtime and cross-kind handles.
   19 struct PhaseHandle {
   20     std::uint64_t value = invalid_graph_handle;
   21 
   22     constexpr PhaseHandle() noexcept = default;
   23     explicit constexpr PhaseHandle(std::uint64_t encoded) noexcept
   24         : value(encoded) {}
   25     constexpr PhaseHandle(
   26         std::uint32_t owner,
   27         std::uint32_t index) noexcept
   28         : value(
   29               (static_cast<std::uint64_t>(owner) << 32u) |
   30               static_cast<std::uint64_t>(index)) {}
   31 
   32     [[nodiscard]] constexpr bool valid() const noexcept {
   33         return value != invalid_graph_handle &&
   34                (static_cast<std::uint32_t>(value) &
   35                 (graph_resource_kind_bit | graph_rate_domain_kind_bit)) == 0;
   36     }
   37     [[nodiscard]] constexpr std::uint32_t owner() const noexcept {
   38         return static_cast<std::uint32_t>(value >> 32u);
   39     }
   40     [[nodiscard]] constexpr std::uint32_t index() const noexcept {
   41         return static_cast<std::uint32_t>(value) &
   42                graph_handle_index_mask;
   43     }
   44 
   45     friend constexpr bool operator==(
   46         PhaseHandle,
   47         PhaseHandle) noexcept = default;
