   53 /*
   54  * Blocking transfer belongs exclusively to fixed backend I/O workers.
   55  * submit() and poll() never call it. The production adapter implements this
   56  * surface with pread()/pwrite() on the official Xilinx XDMA AXI-MM character
   57  * devices; tests inject a driver without needing hardware or kernel headers.
   58  */
   59 struct XdmaDriverApi {
   60     /* The default and the complete positional prefix remain version 1. */
   61     std::uint32_t struct_size = xdma_driver_api_v1_size;
   62     std::uint32_t api_version = xdma_driver_api_version;
   63     void* user_data = nullptr;
   64 
   65     /*
   66      * A non-success initialize result may still represent partial driver
   67      * ownership. The backend makes one immediate shutdown rollback attempt;
   68      * shutdown must therefore be safe after any initialize call. A shutdown
   69      * result of success or invalid_value proves that no ownership remains.
   70      * Any other result must remain retryable by a later shutdown call.
   71      */
   72     XdmaDriverResult (*initialize)(void* user_data) noexcept = nullptr;
   73     XdmaTransferResult (*transfer)(
   74         void* user_data,
   75         XdmaDirection direction,
   76         std::uint32_t channel,
   77         std::uint64_t device_offset,
   78         void* host_data,
   79         std::uint64_t bytes) noexcept = nullptr;
   80     XdmaDriverResult (*reset)(void* user_data) noexcept = nullptr;
   81     XdmaDriverResult (*shutdown)(void* user_data) noexcept = nullptr;
   82     std::uint64_t (*monotonic_time_ns)(void* user_data) noexcept = nullptr;
   83 
   84     std::uint64_t reserved[8]{};
   85 
   86     /* Version-2 tail. Version 1 ignores an entirely zero tail. */
   87     XdmaControlReadResult (*control_read32)(
   88         void* user_data,
   89         std::uint32_t offset) noexcept = nullptr;
   90     XdmaDriverResult (*control_write32)(
   91         void* user_data,
   92         std::uint32_t offset,
   93         std::uint32_t value) noexcept = nullptr;
   94     XdmaUserEventResult (*wait_user_event)(
   95         void* user_data,
   96         std::uint32_t event_index,
   97         std::uint64_t timeout_ns) noexcept = nullptr;
   98     XdmaDriverResult (*request_stop)(void* user_data) noexcept = nullptr;
   99     std::uint64_t reserved_v2[4]{};
  100 };

  149 [[nodiscard]] inline bool set_xdma_control_read(
  150     DeviceCommand& command,
  151     std::uint32_t offset,
  152     const HalV2BufferReference& output) noexcept {
  153     if ((offset & 3u) != 0 || offset >= xdma_control_aperture_max_bytes) {
  154         return false;
  155     }
  156     command = {};
  157     command.kind = static_cast<std::uint32_t>(HalV2CommandKind::dispatch);
  158     command.opcode = xdma_device_opcode_control_read_base | (offset / 4u);
  159     command.buffer_count = 1;
  160     command.buffers[0] = output;
  161     return true;
  162 }
  163 
  164 [[nodiscard]] inline bool set_xdma_control_write(
  165     DeviceCommand& command,
  166     std::uint32_t offset,
  167     std::uint32_t value) noexcept {
  168     if ((offset & 3u) != 0 || offset >= xdma_control_aperture_max_bytes) {
  169         return false;
  170     }
  171     command = {};
  172     command.kind = static_cast<std::uint32_t>(HalV2CommandKind::dispatch);
  173     command.opcode = xdma_device_opcode_control_write_base | (offset / 4u);
  174     command.payload_size = sizeof(value);
  175     command.payload[0] = static_cast<std::uint8_t>(value);
  176     command.payload[1] = static_cast<std::uint8_t>(value >> 8u);
  177     command.payload[2] = static_cast<std::uint8_t>(value >> 16u);
  178     command.payload[3] = static_cast<std::uint8_t>(value >> 24u);
  179     return true;
  180 }

  182 [[nodiscard]] inline bool set_xdma_user_event_wait(
  183     DeviceCommand& command,
  184     std::uint32_t event_index,
  185     const HalV2BufferReference& output) noexcept {
  186     if (event_index >= xdma_user_event_capacity) {
  187         return false;
  188     }
  189     command = {};
  190     command.kind = static_cast<std::uint32_t>(HalV2CommandKind::dispatch);
  191     command.opcode = xdma_device_opcode_user_event_base | event_index;
  192     command.buffer_count = 1;
  193     command.buffers[0] = output;
  194     return true;
  195 }
