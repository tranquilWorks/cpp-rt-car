   11 // Generic quantity wrapper enforcing dimensional correctness.
   12 template <typename Unit, typename Rep = double>
   13 struct quantity {
   14     Rep value;
   15     constexpr explicit quantity(Rep v = {}) : value(v) {}
   16     constexpr Rep count() const { return value; }
   17 };
