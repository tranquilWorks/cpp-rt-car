cmake_minimum_required(VERSION 3.20)
project(rtfw_sdk_recipes LANGUAGES C CXX)
set(RECIPE_RTFW_SOURCE "" CACHE PATH "RTFW source checkout for embedding")
option(RECIPE_WITH_BENCHMARK "Exercise the optional benchmark component" OFF)
if(RECIPE_RTFW_SOURCE AND NOT TARGET rtfw::runtime)
    if(NOT EXISTS "${RECIPE_RTFW_SOURCE}/CMakeLists.txt")
        message(FATAL_ERROR "RECIPE_RTFW_SOURCE must name an RTFW source checkout")
    endif()
    add_subdirectory("${RECIPE_RTFW_SOURCE}" rtfw-build)
elseif(NOT TARGET rtfw::runtime)
    set(recipe_components runtime c_static cuda_backend xdma_backend)
    if(RECIPE_WITH_BENCHMARK)
        list(APPEND recipe_components benchmark)
    endif()
    find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS ${recipe_components})
endif()
# Preserve the original consumer/support sources, including their relative include.
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/package_consumer/c_consumer.c")
    set(recipe_sources "${CMAKE_CURRENT_LIST_DIR}/package_consumer")
else()
    set(recipe_sources "${CMAKE_CURRENT_LIST_DIR}/../../tests/package_consumer")
endif()
function(rtfw_recipe name file library)
    add_executable(recipe_${name} "${recipe_sources}/${file}")
    target_link_libraries(recipe_${name} PRIVATE ${library})
    add_test(NAME recipe_${name} COMMAND recipe_${name})
    set_tests_properties(recipe_${name} PROPERTIES TIMEOUT 120)
endfunction()
enable_testing()
rtfw_recipe(raw_c c_consumer.c rtfw::c_static)
rtfw_recipe(raw_cpp cpp_consumer.cpp rtfw::runtime)
rtfw_recipe(profile profile_consumer.cpp rtfw::runtime)
rtfw_recipe(mixed_rate mixed_rate_consumer.cpp rtfw::runtime)
rtfw_recipe(live_control live_control_consumer.cpp rtfw::runtime)
rtfw_recipe(replay live_control_replay_consumer.cpp rtfw::runtime)
add_test(NAME recipe_replay_replaced COMMAND recipe_replay --replaced)
add_test(NAME recipe_replay_rejected COMMAND recipe_replay --rejected)
set_tests_properties(recipe_replay_replaced recipe_replay_rejected PROPERTIES TIMEOUT 120)
rtfw_recipe(cuda_boundary cuda_consumer.cpp rtfw::cuda_backend)
rtfw_recipe(xdma_boundary xdma_consumer.cpp rtfw::xdma_backend)
if(RECIPE_WITH_BENCHMARK)
    rtfw_recipe(benchmark benchmark_consumer.cpp rtfw::benchmark)
endif()
