cmake_minimum_required(VERSION 3.20)
project(host_adapter LANGUAGES CXX)
set(HOST_RTFW_SOURCE "" CACHE PATH "RTFW source checkout for embedding")
if(HOST_RTFW_SOURCE)
    if(NOT EXISTS "${HOST_RTFW_SOURCE}/CMakeLists.txt")
        message(FATAL_ERROR "HOST_RTFW_SOURCE must name an RTFW source checkout")
    endif()
    add_subdirectory("${HOST_RTFW_SOURCE}" rtfw-build)
elseif(NOT TARGET rtfw::runtime)
    find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS runtime)
endif()
if(NOT (WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux") OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Host reference requires 64-bit Linux/Windows with 4096-byte OS pages")
endif()
add_executable(host_adapter main.cpp)
target_link_libraries(host_adapter PRIVATE rtfw::runtime)
if(WIN32)
    target_link_libraries(host_adapter PRIVATE psapi)
endif()
enable_testing()
foreach(mode native host)
    add_test(NAME host_adapter_${mode} COMMAND ${CMAKE_COMMAND}
        -DHOST_EXECUTABLE=$<TARGET_FILE:host_adapter> -DHOST_MODE=${mode}
        -P "${CMAKE_CURRENT_SOURCE_DIR}/check_output.cmake")
    set_tests_properties(host_adapter_${mode} PROPERTIES TIMEOUT 40)
endforeach()
