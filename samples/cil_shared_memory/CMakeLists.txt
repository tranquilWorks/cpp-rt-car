cmake_minimum_required(VERSION 3.20)
project(cil_shared_memory LANGUAGES CXX)
set(CIL_RTFW_SOURCE "" CACHE PATH "RTFW source checkout for add_subdirectory")
if(CIL_RTFW_SOURCE)
    if(NOT EXISTS "${CIL_RTFW_SOURCE}/CMakeLists.txt")
        message(FATAL_ERROR "CIL_RTFW_SOURCE must name an RTFW source checkout")
    endif()
    add_subdirectory("${CIL_RTFW_SOURCE}" rtfw-build)
elseif(NOT TARGET rtfw::runtime)
    find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS runtime)
endif()
if(NOT (WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux") OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8
   OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    message(FATAL_ERROR "CIL source reference supports Linux/Windows little-endian x86-64 only")
endif()
foreach(role plant controller)
    add_executable(cil_${role} ${role}.cpp)
    target_link_libraries(cil_${role} PRIVATE rtfw::runtime)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_libraries(cil_${role} PRIVATE rt)
    endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME cil_shared_memory_processes
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/runner.py"
        --plant "$<TARGET_FILE:cil_plant>" --controller "$<TARGET_FILE:cil_controller>")
set_tests_properties(cil_shared_memory_processes PROPERTIES TIMEOUT 90)
