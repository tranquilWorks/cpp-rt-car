cmake_minimum_required(VERSION 3.20)
project(golden_system LANGUAGES CXX)
set(GOLDEN_RTFW_SOURCE "" CACHE PATH "RTFW source checkout for embedding")
if(GOLDEN_RTFW_SOURCE)
    if(NOT EXISTS "${GOLDEN_RTFW_SOURCE}/CMakeLists.txt")
        message(FATAL_ERROR "GOLDEN_RTFW_SOURCE must name an RTFW source checkout")
    endif()
    add_subdirectory("${GOLDEN_RTFW_SOURCE}" rtfw-build)
elseif(NOT TARGET rtfw::runtime)
    find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS runtime)
endif()
if(NOT (WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux") OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Golden source kit requires 64-bit Linux/Windows")
endif()
foreach(name system controller)
    if(name STREQUAL "system")
        set(source main.cpp)
    else()
        set(source controller.cpp)
    endif()
    add_executable(golden_${name} ${source})
    target_link_libraries(golden_${name} PRIVATE rtfw::runtime)
    if(WIN32)
        target_link_libraries(golden_${name} PRIVATE psapi)
    elseif(UNIX)
        target_link_libraries(golden_${name} PRIVATE rt)
    endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/provenance.json")
    set(golden_provenance "${CMAKE_CURRENT_SOURCE_DIR}/provenance.json")
elseif(GOLDEN_RTFW_SOURCE)
    set(golden_provenance "${CMAKE_CURRENT_BINARY_DIR}/rtfw-build/golden-provenance.json")
else()
    message(FATAL_ERROR "Golden source provenance is missing")
endif()
enable_testing()
foreach(mode native host external)
    add_test(NAME golden_kit_${mode} COMMAND ${Python3_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/run.py" --plant $<TARGET_FILE:golden_system>
        --controller $<TARGET_FILE:golden_controller> --mode ${mode}
        --provenance "${golden_provenance}" --output "${CMAKE_CURRENT_BINARY_DIR}/${mode} evidence")
    set_tests_properties(golden_kit_${mode} PROPERTIES TIMEOUT 60)
endforeach()
