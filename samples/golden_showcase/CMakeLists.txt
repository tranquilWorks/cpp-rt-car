cmake_minimum_required(VERSION 3.20)
project(golden_showcase LANGUAGES CXX)
set(GOLDEN_RTFW_SOURCE "" CACHE PATH "RTFW source checkout for embedding")
if(GOLDEN_RTFW_SOURCE)
    if(NOT EXISTS "${GOLDEN_RTFW_SOURCE}/CMakeLists.txt")
        message(FATAL_ERROR "GOLDEN_RTFW_SOURCE must name an RTFW source checkout")
    endif()
    set(RTFW_BUILD_BENCHMARKS ON CACHE BOOL "Build optional benchmark component")
    add_subdirectory("${GOLDEN_RTFW_SOURCE}" rtfw-build)
elseif(NOT TARGET rtfw::benchmark)
    find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS runtime cuda_backend xdma_backend benchmark)
endif()
if(NOT TARGET rtfw::benchmark)
    message(FATAL_ERROR "Golden showcase requires the optional benchmark component")
endif()
if(NOT (WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux") OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Golden showcase requires 64-bit Linux/Windows")
endif()
add_library(golden_showcase_provider STATIC provider.cpp variants.cpp)
target_link_libraries(golden_showcase_provider PUBLIC rtfw::runtime rtfw::benchmark rtfw::cuda_backend rtfw::xdma_backend)
if(WIN32)
    target_link_libraries(golden_showcase_provider PUBLIC psapi)
elseif(UNIX)
    target_link_libraries(golden_showcase_provider PUBLIC rt)
endif()
add_executable(golden_showcase main.cpp)
target_link_libraries(golden_showcase PRIVATE golden_showcase_provider)
add_executable(golden_experiment experiment_main.cpp experiments.cpp)
target_link_libraries(golden_experiment PRIVATE golden_showcase_provider)
add_executable(golden_telemetry_loss telemetry_loss.cpp)
target_link_libraries(golden_telemetry_loss PRIVATE golden_showcase_provider)
foreach(kit golden_system golden_cuda golden_xdma)
    add_executable(showcase_${kit} ../${kit}/main.cpp)
    set_target_properties(showcase_${kit} PROPERTIES OUTPUT_NAME ${kit})
    target_link_libraries(showcase_${kit} PRIVATE golden_showcase_provider)
endforeach()
add_executable(showcase_golden_controller ../golden_system/controller.cpp)
set_target_properties(showcase_golden_controller PROPERTIES OUTPUT_NAME golden_controller)
target_link_libraries(showcase_golden_controller PRIVATE golden_showcase_provider)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/provenance.json")
    set(golden_showcase_provenance "${CMAKE_CURRENT_SOURCE_DIR}/provenance.json")
elseif(GOLDEN_RTFW_SOURCE)
    set(golden_showcase_provenance "${CMAKE_CURRENT_BINARY_DIR}/rtfw-build/golden-showcase-provenance.json")
else()
    message(FATAL_ERROR "Golden showcase source provenance is missing")
endif()
enable_testing()
add_test(NAME golden_showcase_kit COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/test.py"
    --build "${CMAKE_CURRENT_BINARY_DIR}" --provenance "${golden_showcase_provenance}")
set_tests_properties(golden_showcase_kit PROPERTIES TIMEOUT 300)
