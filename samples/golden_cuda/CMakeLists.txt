cmake_minimum_required(VERSION 3.20)
project(golden_cuda LANGUAGES CXX)
set(GOLDEN_RTFW_SOURCE "" CACHE PATH "RTFW source checkout for embedding")
option(GOLDEN_CUDA_REAL "Build optional native CUDA host and integer kernel" OFF)
if(GOLDEN_RTFW_SOURCE)
    if(NOT EXISTS "${GOLDEN_RTFW_SOURCE}/CMakeLists.txt")
        message(FATAL_ERROR "GOLDEN_RTFW_SOURCE must name an RTFW source checkout")
    endif()
    add_subdirectory("${GOLDEN_RTFW_SOURCE}" rtfw-build)
elseif(NOT TARGET rtfw::runtime)
    set(components runtime cuda_backend)
    if(GOLDEN_CUDA_REAL)
        list(APPEND components cuda_driver)
    endif()
    find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS ${components})
endif()
if(NOT (WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux") OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Golden CUDA requires 64-bit Linux/Windows")
endif()
add_executable(golden_cuda main.cpp)
target_link_libraries(golden_cuda PRIVATE rtfw::runtime rtfw::cuda_backend)
if(WIN32)
    target_link_libraries(golden_cuda PRIVATE psapi)
elseif(UNIX)
    target_link_libraries(golden_cuda PRIVATE rt)
endif()
if(NOT TARGET golden_controller)
    add_executable(golden_controller ../golden_system/controller.cpp)
    target_link_libraries(golden_controller PRIVATE rtfw::runtime)
    if(UNIX)
        target_link_libraries(golden_controller PRIVATE rt)
    endif()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/provenance.json")
    set(golden_cuda_provenance "${CMAKE_CURRENT_SOURCE_DIR}/provenance.json")
elseif(GOLDEN_RTFW_SOURCE)
    set(golden_cuda_provenance "${CMAKE_CURRENT_BINARY_DIR}/rtfw-build/golden-cuda-provenance.json")
else()
    message(FATAL_ERROR "Golden CUDA source provenance is missing")
endif()
enable_testing()
foreach(mode native host external)
    foreach(dispatch cpu kernel graph)
        add_test(NAME golden_cuda_kit_${mode}_${dispatch} COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/run.py" --plant $<TARGET_FILE:golden_cuda>
            --controller $<TARGET_FILE:golden_controller> --mode ${mode} --dispatch ${dispatch}
            --provenance "${golden_cuda_provenance}" --output "${CMAKE_CURRENT_BINARY_DIR}/${mode}-${dispatch} evidence")
        set_tests_properties(golden_cuda_kit_${mode}_${dispatch} PROPERTIES TIMEOUT 60)
    endforeach()
endforeach()
if(GOLDEN_CUDA_REAL)
    include("${CMAKE_CURRENT_SOURCE_DIR}/real.cmake")
    rtfw_add_golden_cuda_real(golden_cuda_real "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
