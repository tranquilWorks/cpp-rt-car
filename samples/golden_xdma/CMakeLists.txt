cmake_minimum_required(VERSION 3.20)
project(golden_xdma LANGUAGES CXX)
set(GOLDEN_RTFW_SOURCE "" CACHE PATH "RTFW source checkout for embedding")
if(GOLDEN_RTFW_SOURCE)
    if(NOT EXISTS "${GOLDEN_RTFW_SOURCE}/CMakeLists.txt")
        message(FATAL_ERROR "GOLDEN_RTFW_SOURCE must name an RTFW source checkout")
    endif()
    add_subdirectory("${GOLDEN_RTFW_SOURCE}" rtfw-build)
elseif(NOT TARGET rtfw::runtime)
    set(components runtime cuda_backend xdma_backend)
    find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS ${components})
endif()
if(NOT (WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux") OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Golden XDMA requires 64-bit Linux/Windows")
endif()
add_executable(golden_xdma main.cpp)
target_link_libraries(golden_xdma PRIVATE rtfw::runtime rtfw::cuda_backend rtfw::xdma_backend)
if(WIN32)
    target_link_libraries(golden_xdma PRIVATE psapi)
elseif(UNIX)
    target_link_libraries(golden_xdma PRIVATE rt)
endif()
if(NOT TARGET golden_controller)
    add_executable(golden_controller ../golden_system/controller.cpp)
    target_link_libraries(golden_controller PRIVATE rtfw::runtime)
    if(UNIX)
        target_link_libraries(golden_controller PRIVATE rt)
    endif()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/provenance.json")
    set(golden_xdma_provenance "${CMAKE_CURRENT_SOURCE_DIR}/provenance.json")
elseif(GOLDEN_RTFW_SOURCE)
    set(golden_xdma_provenance "${CMAKE_CURRENT_BINARY_DIR}/rtfw-build/golden-xdma-provenance.json")
else()
    message(FATAL_ERROR "Golden XDMA source provenance is missing")
endif()
enable_testing()
foreach(mode native host external)
    foreach(dispatch cpu kernel graph)
        add_test(NAME golden_xdma_kit_${mode}_${dispatch} COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/run.py" --plant $<TARGET_FILE:golden_xdma>
            --controller $<TARGET_FILE:golden_controller> --mode ${mode} --dispatch ${dispatch}
            --provenance "${golden_xdma_provenance}" --output "${CMAKE_CURRENT_BINARY_DIR}/${mode}-${dispatch} evidence")
        set_tests_properties(golden_xdma_kit_${mode}_${dispatch} PROPERTIES TIMEOUT 60)
    endforeach()
endforeach()
