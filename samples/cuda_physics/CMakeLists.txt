cmake_minimum_required(VERSION 3.20)
project(rtfw_cuda_physics_consumer LANGUAGES CXX)
option(PHYSICS_REAL_CUDA "Compile the explicitly opt-in CUDA host and kernel" OFF)
set(PHYSICS_RTFW_SOURCE "" CACHE PATH "Optional source checkout for add_subdirectory embedding")
if(PHYSICS_RTFW_SOURCE)
  set(RTFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(RTFW_BUILD_EXPERIMENTAL OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTS OFF CACHE BOOL "" FORCE)
  set(RTFW_ENABLE_CUDA ${PHYSICS_REAL_CUDA} CACHE BOOL "" FORCE)
  add_subdirectory("${PHYSICS_RTFW_SOURCE}" rtfw)
else()
  if(PHYSICS_REAL_CUDA)
    find_package(rtfw 1.2.1 CONFIG REQUIRED COMPONENTS runtime cuda_backend cuda_driver)
  else()
    find_package(rtfw 1.2.1 CONFIG REQUIRED COMPONENTS runtime cuda_backend)
  endif()
endif()
add_executable(physics_consumer main.cpp)
target_link_libraries(physics_consumer PRIVATE rtfw::runtime rtfw::cuda_backend)
enable_testing()
add_test(NAME physics_consumer COMMAND physics_consumer --count 17 --steps 7 --seed 4294967295)
if(PHYSICS_REAL_CUDA)
  include(real.cmake)
  rtfw_add_physics_real(physics_real_consumer "${CMAKE_CURRENT_SOURCE_DIR}")
endif()

add_executable(pipeline_consumer pipeline/main.cpp)
target_link_libraries(pipeline_consumer PRIVATE rtfw::runtime rtfw::cuda_backend)
foreach(dispatch kernel graph)
  foreach(schedule frame active)
    add_test(NAME pipeline_${dispatch}_${schedule} COMMAND pipeline_consumer
      --dispatch ${dispatch} --schedule ${schedule} --count 17 --steps 7 --seed 4294967295)
  endforeach()
endforeach()
if(PHYSICS_REAL_CUDA)
  rtfw_add_physics_real(pipeline_real_consumer "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/pipeline/real.cpp")
endif()

add_executable(lifetime_consumer lifetime/main.cpp)
target_link_libraries(lifetime_consumer PRIVATE rtfw::runtime rtfw::cuda_backend)
add_test(NAME lifetime_consumer COMMAND lifetime_consumer)
set_tests_properties(lifetime_consumer PROPERTIES TIMEOUT 120)
