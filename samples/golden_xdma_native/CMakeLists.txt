cmake_minimum_required(VERSION 3.20)
project(golden_xdma_native LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Native golden XDMA preparation requires 64-bit Linux")
endif()
if(NOT TARGET rtfw::xdma_linux)
  find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS runtime xdma_backend xdma_linux)
endif()
add_executable(golden_xdma_native main.cpp)
target_link_libraries(golden_xdma_native PRIVATE rtfw::runtime rtfw::xdma_linux rt)
