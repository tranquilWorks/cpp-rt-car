cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(rtfw_closeout_startup LANGUAGES CXX)
  enable_testing()
  find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS runtime xdma_backend)
endif()
add_executable(rtfw_closeout_startup main.cpp ../golden_cuda/allocation.cpp)
target_link_libraries(rtfw_closeout_startup PRIVATE rtfw::runtime rtfw::xdma_backend)
target_compile_features(rtfw_closeout_startup PRIVATE cxx_std_20)
if(COMMAND rtfw_enable_project_warnings)
  rtfw_enable_project_warnings(rtfw_closeout_startup)
elseif(MSVC)
  target_compile_options(rtfw_closeout_startup PRIVATE /W4 /WX)
else()
  target_compile_options(rtfw_closeout_startup PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
add_test(NAME closeout_startup COMMAND rtfw_closeout_startup)
set_tests_properties(closeout_startup PROPERTIES TIMEOUT 60)
