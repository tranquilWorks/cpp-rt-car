cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(rtfw_replay_rejection_consumer LANGUAGES CXX)
    enable_testing()
    find_package(rtfw CONFIG REQUIRED)
endif()

add_executable(rtfw_replay_rejection_tests main.cpp ../golden_cuda/allocation.cpp)
target_link_libraries(rtfw_replay_rejection_tests PRIVATE rtfw::runtime)
target_compile_features(rtfw_replay_rejection_tests PRIVATE cxx_std_20)
if(COMMAND rtfw_enable_project_warnings)
    rtfw_enable_project_warnings(rtfw_replay_rejection_tests)
elseif(MSVC)
    target_compile_options(rtfw_replay_rejection_tests PRIVATE /W4 /WX)
else()
    target_compile_options(rtfw_replay_rejection_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
add_test(NAME host_adapter_replay_rejection COMMAND rtfw_replay_rejection_tests)
set_tests_properties(host_adapter_replay_rejection PROPERTIES TIMEOUT 60)
if(TARGET rtfw_host_adapter_tests)
    add_dependencies(rtfw_host_adapter_tests rtfw_replay_rejection_tests)
endif()

# The standalone default SDK consumer needs only the public Runtime target.
# The optional SDK also supplies the unchanged full golden showcase source kit.
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(RTFW_REPLAY_GOLDEN_KIT "${CMAKE_CURRENT_SOURCE_DIR}/../../samples/golden_showcase")
endif()
if(RTFW_REPLAY_GOLDEN_KIT)
    if(NOT EXISTS "${RTFW_REPLAY_GOLDEN_KIT}/session.hpp")
        message(FATAL_ERROR "Missing golden showcase source kit")
    endif()
    add_executable(rtfw_replay_rejection_golden golden_probe.cpp)
    target_include_directories(rtfw_replay_rejection_golden PRIVATE "${RTFW_REPLAY_GOLDEN_KIT}")
    target_link_libraries(rtfw_replay_rejection_golden PRIVATE rtfw::runtime)
    target_compile_features(rtfw_replay_rejection_golden PRIVATE cxx_std_20)
    if(COMMAND rtfw_enable_project_warnings)
        rtfw_enable_project_warnings(rtfw_replay_rejection_golden)
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/golden-recovery")
    add_test(NAME host_adapter_replay_rejection_golden COMMAND rtfw_replay_rejection_golden
        "${CMAKE_CURRENT_BINARY_DIR}/golden-recovery/result.json")
    set_tests_properties(host_adapter_replay_rejection_golden PROPERTIES TIMEOUT 60)
    if(TARGET rtfw_host_adapter_tests)
        add_dependencies(rtfw_host_adapter_tests rtfw_replay_rejection_golden)
    endif()
endif()
