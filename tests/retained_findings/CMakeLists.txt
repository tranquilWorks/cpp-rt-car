cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(rtfw_retained_findings LANGUAGES CXX)
  enable_testing()
  find_package(rtfw CONFIG REQUIRED COMPONENTS cpp_runtime xdma_backend experimental)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
get_filename_component(retained_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(observed "${CMAKE_CURRENT_BINARY_DIR}/observed")
add_custom_command(OUTPUT "${observed}/sampled.hpp" "${observed}/overlap.cpp"
 COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py" "${retained_root}" "${observed}"
 DEPENDS prepare.py "${retained_root}/tests/package_consumer/sampled_io_simulation_consumer.cpp" "${retained_root}/tests/test_gpu_stub.cpp" VERBATIM)
add_executable(rtfw_retained_startup startup.cpp "${observed}/sampled.hpp")
target_include_directories(rtfw_retained_startup PRIVATE "${observed}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(rtfw_retained_startup PRIVATE rtfw::runtime rtfw::xdma_backend Threads::Threads)
foreach(mode mock native gate)
 add_test(NAME host_adapter_retained_startup_${mode} COMMAND rtfw_retained_startup ${mode})
 set_tests_properties(host_adapter_retained_startup_${mode} PROPERTIES TIMEOUT 30)
endforeach()
add_executable(rtfw_retained_fma fma.cpp ../golden_cuda/allocation.cpp)
target_include_directories(rtfw_retained_fma PRIVATE "${retained_root}/rt/include")
target_compile_definitions(rtfw_retained_fma PRIVATE RT_NUMERICS_FORCE_FMA=1)
target_link_libraries(rtfw_retained_fma PRIVATE Threads::Threads)
add_test(NAME host_adapter_retained_fma COMMAND rtfw_retained_fma)
set_tests_properties(host_adapter_retained_fma PROPERTIES TIMEOUT 30)
add_executable(rtfw_retained_replay replay.cpp ../golden_cuda/allocation.cpp)
target_link_libraries(rtfw_retained_replay PRIVATE rtfw::runtime)
add_test(NAME host_adapter_retained_replay COMMAND rtfw_retained_replay)
set_tests_properties(host_adapter_retained_replay PROPERTIES TIMEOUT 60)
if(SIM_GTEST_AVAILABLE AND RTFW_BUILD_EXPERIMENTAL)
 add_executable(rtfw_retained_overlap "${observed}/overlap.cpp")
 target_link_libraries(rtfw_retained_overlap PRIVATE simcore_platform simcore_rt rtfw_experimental ${SIM_GTEST_LIBRARIES})
 add_test(NAME retained_windows_overlap_control COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/expected_overlap.py" $<TARGET_FILE:rtfw_retained_overlap>)
 set_tests_properties(retained_windows_overlap_control PROPERTIES TIMEOUT 30)
endif()
add_test(NAME host_adapter_retained_proposal COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/proposal.py" "${retained_root}")
foreach(target rtfw_retained_startup rtfw_retained_fma rtfw_retained_replay rtfw_retained_overlap)
 if(TARGET ${target})
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(COMMAND rtfw_enable_project_warnings)
   rtfw_enable_project_warnings(${target})
  endif()
 endif()
endforeach()
if(TARGET rtfw_host_adapter_tests)
 add_dependencies(rtfw_host_adapter_tests rtfw_retained_startup rtfw_retained_fma rtfw_retained_replay)
endif()
add_test(NAME host_adapter_retained_overlay COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/overlay.py" "${retained_root}")
