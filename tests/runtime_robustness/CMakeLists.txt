cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(rtfw_runtime_robustness_consumer LANGUAGES CXX)
    enable_testing()
    find_package(rtfw CONFIG REQUIRED)
endif()
add_executable(rtfw_runtime_robustness main.cpp)
target_compile_features(rtfw_runtime_robustness PRIVATE cxx_std_20)
target_link_libraries(rtfw_runtime_robustness PRIVATE rtfw::runtime)
if(COMMAND rtfw_enable_project_warnings)
    rtfw_enable_project_warnings(rtfw_runtime_robustness)
elseif(MSVC)
    target_compile_options(rtfw_runtime_robustness PRIVATE /W4 /WX)
else()
    target_compile_options(rtfw_runtime_robustness PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
add_test(NAME runtime_robustness COMMAND rtfw_runtime_robustness)
set_tests_properties(runtime_robustness PROPERTIES TIMEOUT 60)
