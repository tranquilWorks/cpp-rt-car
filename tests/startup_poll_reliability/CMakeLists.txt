cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(rtfw_startup_poll_consumer LANGUAGES CXX)
    enable_testing()
    find_package(rtfw CONFIG REQUIRED)
endif()
add_executable(rtfw_startup_poll_tests main.cpp ../golden_cuda/allocation.cpp)
target_link_libraries(rtfw_startup_poll_tests PRIVATE rtfw::runtime rtfw::xdma_backend)
target_compile_features(rtfw_startup_poll_tests PRIVATE cxx_std_20)
if(COMMAND rtfw_enable_project_warnings)
    rtfw_enable_project_warnings(rtfw_startup_poll_tests)
elseif(MSVC)
    target_compile_options(rtfw_startup_poll_tests PRIVATE /W4 /WX)
else()
    target_compile_options(rtfw_startup_poll_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
add_test(NAME host_adapter_startup_poll_reliability COMMAND rtfw_startup_poll_tests)
set_tests_properties(host_adapter_startup_poll_reliability PROPERTIES TIMEOUT 60)
if(TARGET rtfw_host_adapter_tests)
    add_dependencies(rtfw_host_adapter_tests rtfw_startup_poll_tests)
endif()
