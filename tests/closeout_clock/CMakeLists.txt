cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(rtfw_closeout_clock LANGUAGES CXX)
  enable_testing()
  find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS cpp_runtime experimental)
endif()
set(clock_kinds clock)
# Full experimental SimCore currently depends on non-installed HAL headers.
# Keep owner lifecycle coverage in source builds; the relocated SDK exercises
# the repaired clock plus the existing supported public Runtime consumers.
if(TARGET simcore_platform)
  list(APPEND clock_kinds owners)
endif()
foreach(kind IN LISTS clock_kinds)
  add_executable(rtfw_closeout_${kind} ${kind}.cpp)
  target_compile_features(rtfw_closeout_${kind} PRIVATE cxx_std_20)
  target_link_libraries(rtfw_closeout_${kind} PRIVATE rtfw::simcore_rt)
  if(kind STREQUAL "owners")
    if(TARGET simcore_platform)
      target_link_libraries(rtfw_closeout_owners PRIVATE simcore_platform)
    else()
      target_link_libraries(rtfw_closeout_owners PRIVATE rtfw::simcore_platform)
    endif()
  endif()
  if(WIN32)
    target_link_libraries(rtfw_closeout_${kind} PRIVATE psapi)
  endif()
  if(COMMAND rtfw_enable_project_warnings)
    rtfw_enable_project_warnings(rtfw_closeout_${kind})
  endif()
  add_test(NAME closeout_${kind} COMMAND rtfw_closeout_${kind})
  set_tests_properties(closeout_${kind} PROPERTIES TIMEOUT 60)
endforeach()
