cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(rtfw_repair_closeout LANGUAGES CXX)
  enable_testing()
  find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS cpp_runtime experimental)
endif()
find_package(Threads REQUIRED)
add_executable(rtfw_repair_public public.cpp)
target_compile_features(rtfw_repair_public PRIVATE cxx_std_20)
target_link_libraries(rtfw_repair_public PRIVATE rtfw::runtime Threads::Threads)
add_test(NAME host_adapter_repair_public COMMAND rtfw_repair_public)
set_tests_properties(host_adapter_repair_public PROPERTIES TIMEOUT 60)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(observed "${CMAKE_CURRENT_BINARY_DIR}/observed")
  add_custom_command(OUTPUT "${observed}/observed.cpp" "${observed}/observed.hpp"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py"
      --source-root "${CMAKE_SOURCE_DIR}" --output "${observed}"
    DEPENDS prepare.py "${CMAKE_SOURCE_DIR}/rt/src/loopback_backend.cpp"
      "${CMAKE_SOURCE_DIR}/rt/include/rt/loopback_backend.hpp" VERBATIM)
  add_executable(rtfw_repair_ownership ownership.cpp "${observed}/observed.cpp")
  target_compile_features(rtfw_repair_ownership PRIVATE cxx_std_20)
  target_include_directories(rtfw_repair_ownership PRIVATE "${observed}" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(rtfw_repair_ownership PRIVATE rtfw::runtime Threads::Threads)
  foreach(kind submit poll)
    add_test(NAME host_adapter_repair_${kind} COMMAND rtfw_repair_ownership ${kind})
    set_tests_properties(host_adapter_repair_${kind} PROPERTIES TIMEOUT 30)
  endforeach()
  if(TARGET rtfw_host_adapter_tests)
    add_dependencies(rtfw_host_adapter_tests rtfw_repair_public rtfw_repair_ownership)
  endif()
endif()
if(TARGET simcore_platform OR TARGET rtfw::simcore_platform)
  add_executable(rtfw_repair_owners owners.cpp)
  target_compile_features(rtfw_repair_owners PRIVATE cxx_std_20)
  target_link_libraries(rtfw_repair_owners PRIVATE rtfw::simcore_rt)
  if(TARGET simcore_platform)
    target_link_libraries(rtfw_repair_owners PRIVATE simcore_platform)
  else()
    target_link_libraries(rtfw_repair_owners PRIVATE rtfw::simcore_platform)
  endif()
  if(WIN32)
    target_link_libraries(rtfw_repair_owners PRIVATE psapi)
  endif()
  add_test(NAME repair_full_simcore_owners COMMAND rtfw_repair_owners)
  set_tests_properties(repair_full_simcore_owners PROPERTIES TIMEOUT 60)
endif()
foreach(target rtfw_repair_public rtfw_repair_ownership rtfw_repair_owners)
  if(TARGET ${target} AND COMMAND rtfw_enable_project_warnings)
    rtfw_enable_project_warnings(${target})
  endif()
endforeach()
