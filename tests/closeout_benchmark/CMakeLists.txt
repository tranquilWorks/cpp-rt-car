cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(rtfw_closeout_benchmark LANGUAGES CXX)
  enable_testing()
  find_package(rtfw 1.2 CONFIG REQUIRED COMPONENTS runtime benchmark)
  if(NOT RTFW_BENCHMARK_EXAMPLES)
    message(FATAL_ERROR "Set RTFW_BENCHMARK_EXAMPLES to the relocated installed provider directory")
  endif()
else()
  set(RTFW_BENCHMARK_EXAMPLES "${RTFW_PROJECT_SOURCE_DIR}/bench/providers")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(provider "${RTFW_BENCHMARK_EXAMPLES}/runtime_cases")
set(generated "${CMAKE_CURRENT_BINARY_DIR}/device-observed.hpp")
add_custom_command(OUTPUT "${generated}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py"
    --provider-directory "${provider}" --output "${generated}"
  DEPENDS "${provider}/device.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py"
  VERBATIM)
add_executable(rtfw_closeout_benchmark accounting.cpp "${generated}")
target_compile_features(rtfw_closeout_benchmark PRIVATE cxx_std_20)
target_include_directories(rtfw_closeout_benchmark PRIVATE
  "${provider}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(rtfw_closeout_benchmark PRIVATE rtfw::runtime)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  target_link_libraries(rtfw_closeout_benchmark PRIVATE rtfw::benchmark)
else()
  target_include_directories(rtfw_closeout_benchmark PRIVATE "${RTFW_PROJECT_SOURCE_DIR}/bench/include")
endif()
if(COMMAND rtfw_enable_project_warnings)
  rtfw_enable_project_warnings(rtfw_closeout_benchmark)
endif()
add_test(NAME closeout_benchmark COMMAND rtfw_closeout_benchmark)
set_tests_properties(closeout_benchmark PROPERTIES TIMEOUT 120)
