cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(rtfw_telemetry_consumer LANGUAGES CXX)
    enable_testing()
    find_package(rtfw CONFIG REQUIRED COMPONENTS runtime)
    if(NOT TELEMETRY_KIT)
        message(FATAL_ERROR "TELEMETRY_KIT must name the installed source kit")
    endif()
else()
    set(TELEMETRY_KIT "${PROJECT_SOURCE_DIR}/integrations/telemetry")
endif()
add_subdirectory("${TELEMETRY_KIT}" telemetry-kit)
add_executable(rtfw_telemetry_tests main.cpp)
target_link_libraries(rtfw_telemetry_tests PRIVATE rtfw_telemetry_kit)
if(COMMAND rtfw_enable_project_warnings)
    rtfw_enable_project_warnings(rtfw_telemetry_tests)
endif()
add_test(NAME telemetry_queue COMMAND rtfw_telemetry_tests)
set_tests_properties(telemetry_queue PROPERTIES TIMEOUT 60)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME telemetry_spool_contract COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_spool.py" "${TELEMETRY_KIT}/export.py"
    "$<TARGET_FILE:rtfw_telemetry_example>")
set_tests_properties(telemetry_spool_contract PROPERTIES TIMEOUT 60)
if(WIN32)
    add_executable(rtfw_telemetry_etw_tests etw_native.cpp)
    target_link_libraries(rtfw_telemetry_etw_tests PRIVATE rtfw_telemetry_kit tdh)
    add_test(NAME telemetry_etw_native_collection COMMAND rtfw_telemetry_etw_tests)
    set_tests_properties(telemetry_etw_native_collection PROPERTIES TIMEOUT 60)
endif()
