cmake_minimum_required(VERSION 3.20)
project(rtfw_cpu_example LANGUAGES C CXX)
enable_testing()
if (RTFW_SOURCE_DIR)
    set(ENABLE_TESTS OFF CACHE BOOL "" FORCE)
    set(RTFW_BUILD_EXPERIMENTAL OFF CACHE BOOL "" FORCE)
    set(RTFW_BUILD_BENCHMARKS ON CACHE BOOL "" FORCE)
    add_subdirectory("${RTFW_SOURCE_DIR}" rtfw)
    set(cpu_examples "${RTFW_SOURCE_DIR}/bench/providers")
    set(cpu_main "${RTFW_SOURCE_DIR}/tests/package_consumer/benchmark_cpu_consumer.cpp")
else()
    find_package(rtfw CONFIG REQUIRED COMPONENTS runtime benchmark)
    get_filename_component(rtfw_prefix "${rtfw_DIR}/../../.." ABSOLUTE)
    set(cpu_examples "${rtfw_prefix}/share/rtfw/bench/examples")
    set(cpu_main "${cpu_examples}/benchmark_cpu_consumer.cpp")
endif()
add_executable(cpu_example "${cpu_main}" "${cpu_examples}/cpu_provider.cpp")
target_include_directories(cpu_example PRIVATE "${cpu_examples}")
target_link_libraries(cpu_example PRIVATE rtfw::runtime rtfw::benchmark)
add_test(NAME benchmark_cpu_package_consumer COMMAND cpu_example)
