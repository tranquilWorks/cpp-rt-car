cmake_minimum_required(VERSION 3.20)
project(rtfw_device_example LANGUAGES C CXX)
enable_testing()
if(RTFW_SOURCE_DIR)
    file(TO_CMAKE_PATH "${RTFW_SOURCE_DIR}" RTFW_SOURCE_DIR)
    set(ENABLE_TESTS OFF CACHE BOOL "" FORCE)
    set(RTFW_BUILD_EXPERIMENTAL OFF CACHE BOOL "" FORCE)
    set(RTFW_BUILD_BENCHMARKS ON CACHE BOOL "" FORCE)
    add_subdirectory("${RTFW_SOURCE_DIR}" rtfw)
    set(examples "${RTFW_SOURCE_DIR}/bench/providers")
    set(example_main "${RTFW_SOURCE_DIR}/tests/package_consumer/benchmark_device_consumer.cpp")
else()
    find_package(rtfw CONFIG REQUIRED COMPONENTS runtime benchmark cuda_backend xdma_backend)
    get_filename_component(prefix "${rtfw_DIR}/../../.." ABSOLUTE)
    set(examples "${prefix}/share/rtfw/bench/examples")
    set(example_main "${examples}/benchmark_device_consumer.cpp")
endif()
add_executable(device_example "${example_main}" "${examples}/device_provider.cpp")
target_include_directories(device_example PRIVATE "${examples}")
target_link_libraries(device_example PRIVATE rtfw::runtime rtfw::benchmark rtfw::cuda_backend rtfw::xdma_backend)
add_test(NAME benchmark_device_package_consumer COMMAND device_example)
