cmake_minimum_required(VERSION 3.20)
project(benchmark_extension_consumer LANGUAGES C CXX)
if (RTFW_SOURCE_DIR)
    set(RTFW_BUILD_BENCHMARKS ON CACHE BOOL "" FORCE)
    set(RTFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(RTFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(RTFW_BUILD_RUNTIME_DEMO OFF CACHE BOOL "" FORCE)
    set(RTFW_BUILD_EXPERIMENTAL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${RTFW_SOURCE_DIR}" rtfw-build EXCLUDE_FROM_ALL)
    set(extension "${RTFW_SOURCE_DIR}/bench/extension")
else()
    find_package(rtfw CONFIG REQUIRED COMPONENTS benchmark)
    get_filename_component(prefix "${rtfw_DIR}/../../.." ABSOLUTE)
    set(extension "${prefix}/share/rtfw/bench/examples/extension")
endif()
find_package(Threads REQUIRED)
add_executable(extension_lifecycle lifecycle.cpp "${extension}/provider.cpp")
target_include_directories(extension_lifecycle PRIVATE "${extension}")
target_link_libraries(extension_lifecycle PRIVATE rtfw::benchmark Threads::Threads)
enable_testing()
add_test(NAME benchmark_extension_lifecycle COMMAND extension_lifecycle)
