cmake_minimum_required(VERSION 3.20)
project(rtfw_runtime_example LANGUAGES C CXX)
enable_testing()
if(RTFW_SOURCE_DIR)
    file(TO_CMAKE_PATH "${RTFW_SOURCE_DIR}" RTFW_SOURCE_DIR)
    set(ENABLE_TESTS OFF CACHE BOOL "" FORCE)
    set(RTFW_BUILD_EXPERIMENTAL OFF CACHE BOOL "" FORCE)
    set(RTFW_BUILD_BENCHMARKS ON CACHE BOOL "" FORCE)
    add_subdirectory("${RTFW_SOURCE_DIR}" rtfw)
    set(examples "${RTFW_SOURCE_DIR}/bench/providers")
    set(example_main "${RTFW_SOURCE_DIR}/tests/package_consumer/benchmark_runtime_consumer.cpp")
else()
    find_package(rtfw CONFIG REQUIRED COMPONENTS runtime benchmark)
    get_filename_component(prefix "${rtfw_DIR}/../../.." ABSOLUTE)
    set(examples "${prefix}/share/rtfw/bench/examples")
    set(example_main "${examples}/benchmark_runtime_consumer.cpp")
endif()
file(GLOB runtime_cases "${examples}/runtime_cases/*.cpp")
add_executable(runtime_example "${example_main}" "${examples}/runtime_provider.cpp" ${runtime_cases})
target_include_directories(runtime_example PRIVATE "${examples}")
target_link_libraries(runtime_example PRIVATE rtfw::runtime rtfw::benchmark)
add_test(NAME benchmark_runtime_package_consumer COMMAND runtime_example)
